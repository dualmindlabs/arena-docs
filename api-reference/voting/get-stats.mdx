---
title: "Leaderboard"
api: "GET https://api.dualmind.ai/v1/leaderboard"
description: "Retrieve model performance statistics and current ELO rankings."
---

## Authentication

<ParamField header="Authorization" type="string" required>
  ```
  Authorization: Bearer dm_test_xxxxxxxxxxxx
  ```
</ParamField>

## Response

<ResponseField name="items" type="array">
  An ordered list of models by ELO rating.

  <Expandable title="Model statistics object">
    <ResponseField name="modelName" type="string">
      Model identifier (e.g., `llama-3.3-70b`).
    </ResponseField>
    <ResponseField name="displayName" type="string">
      Human-readable model name.
    </ResponseField>
    <ResponseField name="totalVotes" type="integer">
      Total number of comparisons involving this model.
    </ResponseField>
    <ResponseField name="wins" type="integer">
      Number of comparisons where this model was voted the winner.
    </ResponseField>
    <ResponseField name="losses" type="integer">
      Number of comparisons where this model lost.
    </ResponseField>
    <ResponseField name="ties" type="integer">
      Number of tie votes.
    </ResponseField>
    <ResponseField name="winRate" type="number">
      Win percentage as a decimal (e.g., `0.585` = 58.5%).
    </ResponseField>
    <ResponseField name="eloRating" type="number">
      Current ELO rating score. Higher is better.
    </ResponseField>
    <ResponseField name="lastVotedAt" type="string">
      ISO 8601 timestamp of the most recent comparison involving this model.
    </ResponseField>
  </Expandable>
</ResponseField>
